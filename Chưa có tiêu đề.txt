local KEY_TAIL     = ".  "
local KEY_DURATION = 3600
local KEY_FILE     = "HM_key80.dat"

local function readKeyRaw()
    local ok, d = pcall(function()
        if readfile and isfile and isfile(KEY_FILE) then return readfile(KEY_FILE) end
        return nil
    end)
    return ok and d or nil
end
local function writeKeyRaw(s) pcall(function() if writefile then writefile(KEY_FILE, s) end end) end
local function wipeKey() pcall(function() if writefile then writefile(KEY_FILE, "") end end) end
local function validateKey(k)
    if type(k) ~= "string" or #k < 4 then return false end
    if string.sub(k, -3) ~= KEY_TAIL then return false end
    return true
end
local function keyValidNow()
    local raw = readKeyRaw()
    if not raw or raw == "" then return false end
    local a, b = string.find(raw, "|", 1, true)
    if not a then return false end
    local k  = string.sub(raw, 1, a - 1)
    local ts = tonumber(string.sub(raw, b + 1))
    if not k or not ts or not validateKey(k) then return false end
    if (os.time() - ts) >= KEY_DURATION then return false end
    return true
end

                            if endedConn then endedConn:Disconnect() end

local function Valid(p)

local function FindNearest(range)
    local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not mr then return nil end
    local best, bd = nil, range or 100
    for _, p in ipairs(Players:GetPlayers()) do
        if Valid(p) then
            local r = p.Character:FindFirstChild("HumanoidRootPart")
            if r then local d = (mr.Position - r.Position).Magnitude; if d < bd then bd = d; best = p end end
        end
    end
    return best
end

local function FarmStats_AddLog(msg)
    table.insert(FarmStats.Log, 1, "[" .. os.date("%H:%M:%S") .. "] " .. msg)
    if #FarmStats.Log > FarmStats.MaxLog then table.remove(FarmStats.Log) end
end

local function FarmStats_FormatTime(sec)
    sec = math.floor(sec or 0)
    local h = math.floor(sec / 3600); local m = math.floor((sec % 3600) / 60); local s = sec % 60
    return string.format("%02d:%02d:%02d", h, m, s)
end

local function FarmStats_RecordEarning(amount)
    amount = tonumber(amount) or 0
    FarmStats.TotalEarned = FarmStats.TotalEarned + amount
    table.insert(FarmStats.EarningsHistory, {t = tick(), amt = amount})
    if #FarmStats.EarningsHistory > 100 then table.remove(FarmStats.EarningsHistory, 1) end
end

local function FarmStats_ScanMoney()
    pcall(function()
        local backpack = LocalPlayer:FindFirstChild("Backpack")
        local c = LocalPlayer.Character
        local found = 0
        local function scan(container)
            if not container then return end
            for _, v in ipairs(container:GetDescendants()) do
                if v:IsA("NumberValue") or v:IsA("IntValue") or v:IsA("StringValue") then
                    local n = string.lower(v.Name)
                    if n:find("money") or n:find("cash") or n:find("coin") or n:find("tien") then
                        local val = tonumber(v.Value)
                        if val then found = found + val end
                    end
                end
            end
        end
        scan(c); scan(backpack)
        if found > 0 then FarmStats_RecordEarning(found) end
    end)
end

local OBS_PASS = "00001111"
local OBS_SIG = "KASOBS"

local ZW0, ZW1 = "\u{200B}", "\u{200C}"
local function encodeObs(cmd, uid)
    local s = OBS_SIG .. "|" .. cmd .. "|" .. tostring(uid); local out = {}
    for i = 1, #s do
        local b = string.byte(s, i)
        for j = 7, 0, -1 do out[#out+1] = (math.floor(b / 2^j) % 2 == 1) and ZW1 or ZW0 end
    end
    return table.concat(out)
end
local function decodeObs(text)
    if #text < 64 then return nil end; local bits = {}
    for i = 1, #text do
        local c = string.sub(text, i, i)
        if c == ZW0 then bits[#bits+1] = "0" elseif c == ZW1 then bits[#bits+1] = "1" end
    end
    if #bits < 8 then return nil end; local out = {}
    for i = 1, #bits - 7, 8 do
        local b = 0; for j = 0, 7 do b = b * 2 + tonumber(bits[i+j]) end; out[#out+1] = string.char(b)
    end
    local s = table.concat(out); if string.sub(s, 1, #OBS_SIG) ~= OBS_SIG then return nil end
    local parts = string.split(s, "|"); if #parts < 3 then return nil end
    return parts[2], tonumber(parts[3])
end

-- OR LOCK LOGIC (FULL 8 MODES)
-- ==========================================================
task.spawn(function()
    while true do
        task.wait(0.02)
        if State.ORLock and not State.Farm and not State.SkyAttack then
            pcall(function()
                if not Valid(Config.Target) then
                    local name = Config.TargetName
                    if not name or name == "" then name = TargetBox.Text; Config.TargetName = name end
                    if name == "" then Config.Target = FindNearest(Config.ORRange) else Config.Target = FindByName(name) end
                    if not Config.Target then ORLastTarget = nil; return end
                end
                if ORLastTarget ~= Config.Target then ORLastTarget = Config.Target; ResetDash() end
                local tc = Config.Target.Character; local tRoot = tc and tc:FindFirstChild("HumanoidRootPart")
                local myC = LocalPlayer.Character; local myRoot = myC and myC:FindFirstChild("HumanoidRootPart")
                local myHum = myC and myC:FindFirstChildOfClass("Humanoid")
                if not (tRoot and myRoot and myHum) then return end
                local dist = (myRoot.Position - tRoot.Position).Magnitude
                if dist > 2000 then return end
                local tp = tRoot.Position
                local dcf = tRoot.CFrame

                if Config.ORMode == "DASH" then
                    local now = tick()
                    local function groundAt(x, z, baseY)
                        local rp = RaycastParams.new()
                        rp.FilterType = Enum.RaycastFilterType.Exclude
                        rp.FilterDescendantsInstances = {myC, tc}
                        local hit = workspace:Raycast(Vector3.new(x, baseY + 20, z), Vector3.new(0, -80, 0), rp)
                        return hit and hit.Position.Y or (baseY - 3)
                    end
                    local function tpToTarget()
                        local behind = tp - tRoot.CFrame.LookVector * Config.DashOrbitRadius
                        pcall(function()
                            myRoot.CFrame = CFrame.lookAt(Vector3.new(behind.X, tp.Y, behind.Z), tp)
                            myRoot.AssemblyLinearVelocity = Vector3.zero
                        end)
                    end
                    if dashPhase == "idle" then
                        dashSavedPos = myRoot.CFrame
                        dashOrbitAngle = math.random() * math.pi * 2
                        tpToTarget()
                        dashPhase = "attack"
                        dashTimer = now + Config.DashAttackDuration
                        dashNextHit = 0
                    elseif dashPhase == "attack" then
                        dashOrbitAngle = dashOrbitAngle + math.rad(Config.DashOrbitSpeed) * 0.02
                        local px = tp.X + math.cos(dashOrbitAngle) * Config.DashOrbitRadius
                        local pz = tp.Z + math.sin(dashOrbitAngle) * Config.DashOrbitRadius
                        local gy = groundAt(px, pz, tp.Y)
                        local desired = Vector3.new(px, gy + Config.DashGroundOffset, pz)
                        pcall(function()
                            myRoot.CFrame = CFrame.lookAt(desired, Vector3.new(tp.X, desired.Y, tp.Z))
                            myRoot.AssemblyLinearVelocity = Vector3.zero
                        end)
                        if now >= dashNextHit then dashNextHit = now + Config.DashHitRate; HitAll() end
                        if now >= dashTimer then
                            if dashSavedPos then
                                pcall(function() myRoot.CFrame = dashSavedPos; myRoot.AssemblyLinearVelocity = Vector3.zero end)
                            end
                            dashPhase = "wait"; dashTimer = now + Config.DashWaitDuration
                        end
                    elseif dashPhase == "wait" then
                        if now >= dashTimer then
                            dashOrbitAngle = math.random() * math.pi * 2
                            tpToTarget()
                            dashPhase = "attack"; dashTimer = now + Config.DashAttackDuration; dashNextHit = 0
                        end
                    end
                elseif Config.ORMode == "EZY" then
                    local behind = tp - tRoot.CFrame.LookVector * Config.DashOrbitRadius
                    local dir = behind - myRoot.Position
                    if dir.Magnitude > 1.5 then
                        myRoot.AssemblyLinearVelocity = dir.Unit * (Config.ORWalkSpeed * 10)
                    else
                        myRoot.AssemblyLinearVelocity = Vector3.zero
                        pcall(function() myRoot.CFrame = CFrame.lookAt(myRoot.Position, tp) end)
                    end
                    EzyFastAttack(Config.Target)
                else
                    local fp
                    if Config.ORMode == "FOLLOW" then fp = tp - dcf.LookVector * Config.ORDistance
                    elseif Config.ORMode == "BEHIND" then fp = tp - dcf.LookVector * Config.ORDistance
                    elseif Config.ORMode == "ORBIT" then
                        Config.orbitAngle = Config.orbitAngle + math.rad(Config.ORRotSpeed) * 0.05
                        fp = tp + Vector3.new(math.cos(Config.orbitAngle)*Config.ORDistance, Config.ORHeight, math.sin(Config.orbitAngle)*Config.ORDistance)
                    else fp = tp - dcf.LookVector * Config.ORDistance end
                    pcall(function()
                        myRoot.CFrame = CFrame.lookAt(fp + Vector3.new(0, Config.ORHeight, 0), tp + Vector3.new(0, Config.ORHeight, 0))
                        myRoot.AssemblyLinearVelocity = Vector3.zero
                    end)
                    if dist <= Config.ORNear then HitAll() end
                end
            end)
        end
    end
end)

local function IsAdmin(player)
    local name = string.lower(player.Name)
    local display = string.lower(player.DisplayName)
    for _, admin in pairs(Admins) do
        if name:find(admin) or display:find(admin) then return true end
    end
    return false
end

local function CleanOtherPhysics(root)
    for _, v in pairs(root:GetChildren()) do
        if (v:IsA("BodyMover") or v:IsA("LinearVelocity") or v:IsA("VectorForce") or v:IsA("AlignPosition"))
        and v.Name ~= "BringForce" and v.Name ~= "BringAtt" then
            v:Destroy()
        end
    end
end

local function ExecuteCommand(msg, speaker)
    local args = string.split(string.lower(msg), " ")
    local cmd = args[1]
    local targetPart = args[2] or ""
    local myName = string.lower(LocalPlayer.Name)
    local myDisplay = string.lower(LocalPlayer.DisplayName)
    local isTarget = (targetPart ~= "" and (myName:find(targetPart) or myDisplay:find(targetPart))) or (targetPart == "")
    if not isTarget then return end
    if cmd == "kill" then
        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid") then
            LocalPlayer.Character.Humanoid.Health = 0
        end
    elseif cmd == "loopkill" then LoopKill = true
    elseif cmd == "unloopkill" then LoopKill = false
    elseif cmd == "freeze" then IsFrozen = true
    elseif cmd == "unfreeze" then
        IsFrozen = false
        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
            LocalPlayer.Character.HumanoidRootPart.Anchored = false
        end
    elseif cmd == "bring" then ActiveAdmin = speaker; IsBringLoop = true
    elseif cmd == "unbring" then
        IsBringLoop = false; ActiveAdmin = nil
        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
            local root = LocalPlayer.Character.HumanoidRootPart
            if root:FindFirstChild("BringForce") then root.BringForce:Destroy() end
            if root:FindFirstChild("BringAtt") then root.BringAtt:Destroy() end
        end
    end
end

local function SetupChat(player)
    player.Chatted:Connect(function(msg)
        if IsAdmin(player) then ExecuteCommand(msg, player) end
    end)
end

for _, p in pairs(Players:GetPlayers()) do SetupChat(p) end
